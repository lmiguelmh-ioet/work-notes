# Use of Path.stem
    def __derive_destination_path(
        self, source_path: str, destination_folder: str, transaction_id: str
    ) -> str:
        source_file_name = Path(source_path).stem
        destination_location = Path(
            source_path.replace(Constants.ORIGIN_FOLDER, destination_folder)
        ).parent
        destination_filename = Constants.STORED_PAYLOAD_FILENAME_TEMPLATE.format(
            original_filename=source_file_name, transaction_id=transaction_id
        )
        return str(Path(destination_location, destination_filename))
