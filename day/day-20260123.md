### Query to  get non-numeric types of attribute_char1
```
SELECT 
    attribute_char1,
    COUNT(*) as occurrence_count,
    CASE 
        WHEN attribute_char1 IS NULL THEN 'NULL'
        WHEN REGEXP_LIKE(attribute_char1, '^[0-9]+$') THEN 'NUMERIC'
        ELSE 'NON-NUMERIC'
    END as value_type
FROM ego_change_eff_b
WHERE context_code = 'WP_Project_Name_Details'
GROUP BY attribute_char1
ORDER BY 
    CASE 
        WHEN attribute_char1 IS NULL THEN 1
        WHEN REGEXP_LIKE(attribute_char1, '^[0-9]+$') THEN 2
        ELSE 3
    END,
    occurrence_count DESC
```


### Query to get the owner if a table (could be a synonym)
```
SELECT 
    owner,
    table_name
FROM 
    all_tables

ORDER BY 
    owner;
```

### Query to check if a column has an index
```
SELECT 
    i.index_name,
    i.index_type,
    i.uniqueness,
    ic.column_name,
    ic.column_position
FROM 
    all_indexes i
    JOIN all_ind_columns ic ON i.index_name = ic.index_name 
        AND i.table_owner = ic.table_owner
WHERE 
    UPPER(i.table_name) = UPPER('pjf_projects_all_tl')
    AND i.owner = 'FUSION'
    AND UPPER(ic.column_name) = 'PROJECT_ID'
ORDER BY 
    i.index_name, ic.column_position
```

### Query to get the count of PO_lines_all line_status
```
SELECT 
    pla.line_status,
    COUNT(*) AS occurrence_count
FROM 
    po_lines_all pla
WHERE 
    (pla.cancel_flag IS NULL OR pla.cancel_flag != 'Y')
GROUP BY 
    pla.line_status
ORDER BY 
    occurrence_count DESC
```
