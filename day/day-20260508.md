```_create_tracker_action.py
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Optional

from ._tracker import Tracker


@dataclass(frozen=True)
class CreateTrackerRequest:
    """Arguments for starting or resuming tracking with the provider."""

    tracking_number: str
    carrier_hint: Optional[str] = None


class CreateTrackerAction(ABC):
    @abstractmethod
    async def __call__(self, request: CreateTrackerRequest) -> Tracker:
        """
        Start or resume tracking for this number in the provider.

        Semantics: ensure we have a subscription or tracker; idempotency is
        defined per implementation (EasyPost vs USPSS may differ).

        carrier_hint: optional string meaningful only to this provider (EasyPost
        external name, USPSS service code, etc.).
        """
        raise NotImplementedError  # pragma: no cover
```

```_tracker_reader.py
from abc import ABC, abstractmethod
from dataclasses import dataclass
from datetime import datetime
from typing import List, Optional

from ._tracker import Tracker


@dataclass(frozen=True)
class TrackerSearchCriteria:
    """
    Select one lookup shape:

    - By provider tracker id: set ``external_tracker_id`` only (``get_tracker``
      semantics). Implementations return an empty list when the provider
      reports not found.

    - By number and time window: set ``tracking_number``, ``not_before``, and
      ``not_after`` (``find_trackers`` semantics). Matches trackers for that
      number in ``[not_before, not_after]``. Ordering is deterministic for the
      provider (e.g. newest first); callers that only want one take ``first()``
      or apply their own rule.
    """

    external_tracker_id: Optional[str] = None
    tracking_number: Optional[str] = None
    not_before: Optional[datetime] = None
    not_after: Optional[datetime] = None


class TrackerReader(ABC):
    @abstractmethod
    async def __call__(
        self,
        search_criteria: TrackerSearchCriteria,
    ) -> List[Tracker]:
        """
        Fetch trackers according to ``search_criteria``.

        When criteria carry ``external_tracker_id``, behaves like ``get_tracker``:
        at most one hit, empty list if the provider reports not found.

        When criteria carry ``tracking_number`` with ``not_before`` and
        ``not_after``, behaves like ``find_trackers``: zero or more trackers for
        that number in the inclusive window; ordering is deterministic for the
        provider.
        """
        raise NotImplementedError  # pragma: no cover

```

```tracker.py
from dataclasses import dataclass, field
from datetime import datetime
from enum import Enum
from typing import List, Optional


class TrackingStatus(Enum):
    UNKNOWN = "unknown"
    PRE_TRANSIT = "pre_transit"
    IN_TRANSIT = "in_transit"
    OUT_FOR_DELIVERY = "out_for_delivery"
    DELIVERED = "delivered"
    AVAILABLE_FOR_PICKUP = "available_for_pickup"
    RETURN_TO_SENDER = "return_to_sender"
    FAILURE = "failure"
    CANCELLED = "cancelled"
    ERROR = "error"


@dataclass(frozen=True)
class TrackingLocation:
    # EP: tracking_details[].tracking_location.city
    # USPS 3.2 (v3r2): trackingEvents[].eventCity (deprecated Tracking v3 REST: same; legacy Web Tools XML: EventCity)
    city: Optional[str] = None

    # EP: tracking_details[].tracking_location.state
    # USPS 3.2 (v3r2): trackingEvents[].eventState (deprecated v3 REST: same; legacy XML: EventState)
    state: Optional[str] = None

    # EP: tracking_details[].tracking_location.country
    # USPS 3.2 (v3r2): trackingEvents[].eventCountry (deprecated v3 REST: same; legacy XML: EventCountry)
    country: Optional[str] = None

    # EP: tracking_details[].tracking_location.zip
    # USPS 3.2 (v3r2): trackingEvents[].eventZIPCode (deprecated v3 REST: eventZIP; legacy XML: EventZIPCode)
    postal_code: Optional[str] = None


@dataclass(frozen=True)
class TrackingEvent:
    # EP: tracking_details[].datetime (ISO) → parsed instant
    # USPS 3.2 (v3r2): eventTimestamp as AllEventDate (date-only or date-time); GMTTimestamp/GMTOffset optional (deprecated v3 REST: date-time string; legacy XML: EventDate + EventTime)
    occurred_at: Optional[datetime] = None

    # EP: tracking_details[].status (same allowed strings as Tracker.status)
    # USPS 3.2 (v3r2): map trackingEvents[].eventType and/or eventCode / reasonCode / actionCode (Pub 199) → enum (legacy XML: Event)
    status: TrackingStatus = TrackingStatus.UNKNOWN

    # EP: tracking_details[].message
    # USPS 3.2 (v3r2): eventType (legacy XML: Event)
    message: str = ""

    # EP: tracking_details[].description
    # USPS 3.2 (v3r2): no dedicated field on TrackingEvent; use "" or duplicate eventType / derive from reasonCode (3.2 also exposes e.g. eventPartner, acceptanceTypeCategory — not modeled here)
    description: str = ""

    # EP: tracking_details[].source (carrier / feed id; docs examples use "USPS", carrier list includes UspsShip)
    # USPS 3.2 (v3r2): not on payload; adapter constant e.g. USPS_Tracking_v3r2 (deprecated v3: USPS_Tracking_v3; legacy Web Tools XML: USPS_WebTools_TrackV2)
    source: str = ""

    # EP: from tracking_details[].tracking_location
    # USPS 3.2 (v3r2): same TrackingEvent’s eventCity/eventState/eventCountry/eventZIPCode (deprecated v3 REST: eventZIP)
    location: Optional[TrackingLocation] = None


@dataclass(frozen=True)
class Tracker:
    # EP: id (e.g. trk_…)
    # USPS 3.2 (v3r2): synthetic id or SharedTrackingDetail.uniqueTrackingID; stable key e.g. usps:tracking:v3r2:{trackingNumber} (deprecated v3 REST: uniqueMailPieceId optional; legacy XML: TrackInfo@ID)
    external_tracker_id: str = ""

    # EP: tracking_code
    # USPS 3.2 (v3r2): TrackingDetail.trackingNumber (tracking via POST https://apis.usps.com/tracking/v3r2/tracking + TrackingRequest JSON; deprecated v3 REST: GET …/tracking/v3/tracking/{trackingNumber})
    tracking_number: str = ""

    # EP: map each tracking_details[] → TrackingEvent
    # USPS 3.2 (v3r2): TrackingDetail.trackingEvents[] → TrackingEvent (spec order newest-first; normalize to oldest→newest in adapter; deprecated v3 REST: same array semantics; legacy XML: TrackSummary + TrackDetail)
    tracking_events: List[TrackingEvent] = field(default_factory=list)

    # EP: created_at
    # USPS 3.2 (v3r2): adapter-defined e.g. oldest eventTimestamp after reorder; optional SharedTrackingDetail.mailingDate when useful (deprecated v3 REST: mailPieceIntakeDate optional; legacy: first scan / parser-defined)
    created_at: Optional[datetime] = None

    # EP: updated_at
    # USPS 3.2 (v3r2): latest trackingEvents[].eventTimestamp or poll time (same idea on deprecated v3 REST)
    updated_at: Optional[datetime] = None

    # EP: public_url
    # USPS 3.2 (v3r2): not on TrackingDetail schema; None or external consumer URL (deprecated v3 REST / legacy XML: same)
    public_tracking_url: Optional[str] = None

    # EP: est_delivery_date
    # USPS 3.2 (v3r2): deliveryDateExpectation.expectedDeliveryDate / predictedDeliveryDate / guaranteedDelivery* (nested AllGuaranteedDeliveryTimes); deprecated v3 REST TrackingDetails: expectedDeliveryTimeStamp and related flat fields
    estimated_delivery_at: Optional[datetime] = None

    # EP: shipment_id (shp_…) or None
    # USPS 3.2 (v3r2): associatedTrackingNumber when present (not EasyPost shipment); deprecated v3 REST: editedLabelId / associatedLabel when populated; legacy Web Tools: none
    external_shipment_id: Optional[str] = None

    # EP: status on Tracker (normalize string to TrackingStatus)
    # USPS 3.2 (v3r2) SharedTrackingDetail: status or statusCategory → enum; statusSummary is narrative (deprecated v3 REST: same fields on TrackingDetails; legacy XML: StatusSummary / TrackSummary text)
    aggregate_status: Optional[TrackingStatus] = None

```
