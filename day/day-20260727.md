```sql
 -- RMCS-I-3001 - Proof of Delivery for RMCS (FBDI sheet: VRM_SOURCE_DOC_ADDL_SUBLINES)
--
-- Parameters:
--   :P_FROM_DATE              DATE
--   :P_TO_DATE                DATE
--   :P_ORDER_NUMBERS          VARCHAR2 multi-value LOV
--
-- date_source:
--   DELIVERY    -> doo_fulfill_line_details.actual_delivery_date
--   FULFILLMENT -> doo_fulfill_lines_all.fulfillment_date
--
WITH order_fulfillment_cfg AS (
    SELECT 'WP_CUST_ORDER'    AS order_type_code, 'DELIVERY'    AS date_source FROM dual UNION ALL
    SELECT 'WP_CUST_PICKUP'   AS order_type_code, 'DELIVERY'    AS date_source FROM dual UNION ALL
    SELECT 'WP_CUST_TAKEAWAY' AS order_type_code, 'FULFILLMENT' AS date_source FROM dual UNION ALL
    SELECT 'WP_WHOLESALE'     AS order_type_code, 'FULFILLMENT' AS date_source FROM dual
),
fulfill_delivery AS (
    SELECT
        fld.fulfill_line_id,
        MAX(fld.actual_delivery_date) AS actual_delivery_date
    FROM
        doo_fulfill_line_details fld
        INNER JOIN doo_fulfill_lines_all fl
            ON fl.fulfill_line_id = fld.fulfill_line_id
            AND fl.shippable_flag = 'Y'
        INNER JOIN doo_headers_all h
            ON h.header_id = fl.header_id
        INNER JOIN order_fulfillment_cfg cfg
            ON cfg.order_type_code = h.order_type_code
            AND cfg.date_source = 'DELIVERY'
    WHERE
        fld.actual_delivery_date IS NOT NULL
        AND (
            h.source_order_number IN (:P_ORDER_NUMBERS)
            OR (
                :P_FROM_DATE IS NOT NULL
                AND :P_TO_DATE IS NOT NULL
                AND fld.actual_delivery_date >= CAST(:P_FROM_DATE AS DATE)
                AND fld.actual_delivery_date <= CAST(:P_TO_DATE AS DATE)
            )
        )
    GROUP BY
        fld.fulfill_line_id
),
rmcs_lines_base AS (
    SELECT
        fl.fulfill_line_number,
        fl.fulfilled_qty,
        h.source_order_number,
        sdl.doc_line_id_int_1,
        sdl.doc_line_id_int_2,
        sdl.doc_line_id_int_3,
        sdl.doc_line_id_int_4,
        sdl.doc_line_id_int_5,
        sdl.doc_line_id_char_1,
        sdl.doc_line_id_char_2,
        sdl.doc_line_id_char_3,
        sdl.doc_line_id_char_4,
        sdl.doc_line_id_char_5,
        sdl.source_system,
        sdl.document_type_code,
        TO_NUMBER(
            TO_CHAR(pol.customer_contract_header_id)
            || TO_CHAR(pol.document_line_id)
        ) AS doc_addl_sub_line_id_int_1,
        CASE cfg.date_source
            WHEN 'DELIVERY' THEN fld.actual_delivery_date
            WHEN 'FULFILLMENT' THEN fl.fulfillment_date
        END AS fulfillment_date
    FROM
        doo_headers_all h
        INNER JOIN order_fulfillment_cfg cfg
            ON cfg.order_type_code = h.order_type_code
        INNER JOIN doo_lines_all l
            ON l.header_id = h.header_id
        INNER JOIN doo_fulfill_lines_all fl
            ON fl.header_id = h.header_id
            AND fl.line_id = l.line_id
            AND fl.shippable_flag = 'Y'
        LEFT JOIN fulfill_delivery fld
            ON fld.fulfill_line_id = fl.fulfill_line_id
        -- Related source document contract already exists in RMCS
        INNER JOIN vrm_source_doc_lines sdl
            ON sdl.doc_line_id_char_1 = h.source_order_number
            AND sdl.doc_line_id_int_1 = l.line_id
            -- INT_2 may be whole (1) or decimal (1.1) for the same OM line; match on line number.
            AND TRUNC(sdl.doc_line_id_int_2) = l.line_number
            AND sdl.source_system = 'WP_RECEIVABLES'
            AND sdl.document_type_code = 'WP_RECEIVABLES'
        -- Line is on a contract: active performance obligation for that source doc line
        INNER JOIN vrm_perf_obligation_lines pol
            ON pol.document_line_id = sdl.document_line_id
            AND NVL(pol.removed_flag, 'N') = 'N'
    WHERE
        fl.status_code IN (
            'SHIPPED',
            'AWAIT_BILLING',
            'BILLED',
            'CLOSED'
        )
        AND NVL(fl.fulfilled_qty, 0) > 0
        -- Line not already in RMCS as a proof of delivery line
        AND NOT EXISTS (
            SELECT
                1
            FROM
                vrm_source_doc_addl_sublines ads
            WHERE
                ads.document_line_id = sdl.document_line_id
                AND ads.doc_additional_sline_id_int_1 = TO_NUMBER(
                    TO_CHAR(pol.customer_contract_header_id)
                    || TO_CHAR(pol.document_line_id)
                )
                AND ads.source_system = 'WP_RECEIVABLES'
                AND ads.document_type_code = 'WP_RECEIVABLES'
                AND UPPER(ads.additional_se_type_code) LIKE '%PROOF%DELIVERY%'
        )
),
rmcs_lines AS (
    SELECT
        rl.*
    FROM
        rmcs_lines_base rl
    WHERE
        rl.fulfillment_date IS NOT NULL
        AND (
            (
                :P_FROM_DATE IS NOT NULL
                AND :P_TO_DATE IS NOT NULL
                AND rl.fulfillment_date >= CAST(:P_FROM_DATE AS DATE)
                AND rl.fulfillment_date <= CAST(:P_TO_DATE AS DATE)
            )
            OR rl.source_order_number IN (:P_ORDER_NUMBERS)
        )
)
SELECT
    rl.document_type_code,
    rl.source_system,
    rl.doc_addl_sub_line_id_int_1                  AS sub_line_id,
    rl.doc_line_id_int_1                           AS line_id,
    rl.doc_line_id_int_2                           AS line_number,
    rl.doc_line_id_int_3,
    rl.doc_line_id_int_4,
    rl.doc_line_id_int_5,
    rl.doc_line_id_char_1                          AS order_number,
    rl.doc_line_id_char_2,
    rl.doc_line_id_char_3,
    rl.doc_line_id_char_4,
    rl.doc_line_id_char_5,
    rl.fulfilled_qty                               AS satisfied_quantity,
    rl.fulfillment_date                            AS fulfillment_date
FROM
    rmcs_lines rl
ORDER BY
    rl.doc_line_id_char_1,
    rl.doc_line_id_int_2,
    rl.fulfill_line_number

 ```
