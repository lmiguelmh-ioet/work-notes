    async def test__call__when_import_fails__raises_and_moves_to_error(
        self,
        test_data_factory: Callable[..., dict[str, Any]],
        dependencies_factory: Callable[..., dict[str, Any]],
    ) -> None:
        test_data = test_data_factory()
        exception_message = "Oracle ERP import failed"
        dependencies = dependencies_factory(
            file_storage_reader_response=[test_data["payout_file_content"]],
            bank_statement_import_action_response=[Exception(exception_message)],
        )
        listener = WpCmI3004CashReconciliationListener(**dependencies)
        file_path = test_data["file_path"]
        destination_path = file_path.replace(Constants.ORIGIN_FOLDER, Constants.ERROR_FOLDER)

        with pytest.raises(CashReconciliationProcessingError) as exc_info:
            await listener(file_path=file_path, request_id=test_data["request_id"])

        assert exception_message in str(exc_info.value.message)
        assert exc_info.value.integration_name == expected_integration_name
        dependencies["file_storage_action"].move_file.assert_called_once_with(
            origin_path=file_path,
            destination_path=destination_path,
        )
