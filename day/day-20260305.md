```
fa-evdi-test - env

SELECT DISTINCT
    RSH.SHIPMENT_NUM,
    RSH.SHIPMENT_HEADER_ID,
    RSH.CREATION_DATE
FROM
    RCV_SHIPMENT_HEADERS RSH
    INNER JOIN RCV_SHIPMENT_LINES RSL 
        ON RSH.SHIPMENT_HEADER_ID = RSL.SHIPMENT_HEADER_ID
        AND RSL.SHIPMENT_LINE_STATUS_CODE <> 'CANCELLED'
    INNER JOIN PO_HEADERS_ALL PHA 
        ON PHA.PO_HEADER_ID = RSL.PO_HEADER_ID
        AND UPPER(PHA.TYPE_LOOKUP_CODE) = 'STANDARD'
    INNER JOIN POZ_SUPPLIERS_V PSV 
        ON PSV.VENDOR_ID = PHA.VENDOR_ID
    INNER JOIN INV_ORG_PARAMETERS IOP 
        ON IOP.ORGANIZATION_ID = RSL.TO_ORGANIZATION_ID
    INNER JOIN HR_ALL_ORGANIZATION_UNITS_X HOU 
        ON HOU.ORGANIZATION_ID = IOP.ORGANIZATION_ID
        AND UPPER(HOU.NAME) = 'FEDEX SUPPLY CHAIN'
WHERE
    RSH.SHIPMENT_NUM IS NOT NULL
    AND (
        (:P_FROM_DATE IS NOT NULL
            AND :P_TO_DATE IS NOT NULL
            AND RSH.CREATION_DATE >= CAST(:P_FROM_DATE AS DATE)
            AND RSH.CREATION_DATE <= CAST(:P_TO_DATE AS DATE))
        OR (RSH.SHIPMENT_NUM IN (:P_SHIPMENT_NUMBERS))
    )
ORDER BY
    RSH.CREATION_DATE
```
