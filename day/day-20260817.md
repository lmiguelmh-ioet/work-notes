```
_ORACLE_STATUS_TO_DOMAIN: dict[str, DataTransformationStatus] = {
    "PROCESSED": DataTransformationStatus.SUCCESS,
    "REJECTED": DataTransformationStatus.FAILED,
    "INITIALIZED": DataTransformationStatus.PENDING,
    "NOT_PROCESSED": DataTransformationStatus.PENDING,
    "FUSION_INIT": DataTransformationStatus.PENDING,
    "FUSION_NOT_PROCESSED": DataTransformationStatus.PENDING,
    "IGNORED": DataTransformationStatus.EXCLUDED,
    "PURGED": DataTransformationStatus.EXCLUDED,
}

def map_data_transformation_status(oracle_status: str | None) -> DataTransformationStatus:
    if oracle_status is None or not oracle_status.strip():
        return DataTransformationStatus.UNKNOWN

    normalized_status = oracle_status.strip()
    mapped_status = _ORACLE_STATUS_TO_DOMAIN.get(normalized_status)
    if mapped_status is None:
        raise UnknownDataTransformationStatusError(
            f"Unknown DATA_TRANSFORMATION_STATUS '{normalized_status}'"
        )
    return mapped_status

```
