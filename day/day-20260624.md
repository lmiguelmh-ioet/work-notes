-- RMCS-I-3001 - Proof of Delivery for RMCS (FBDI sheet: VRM_SOURCE_DOC_ADDL_SUBLINES)
--
-- Parameters:
--   :P_FROM_DATE              DATE
--   :P_TO_DATE                DATE
--   :P_SALES_ORDER_NUMBERS    VARCHAR2 multi-value LOV
--
WITH fulfill_delivery AS (
    SELECT
        fld.fulfill_line_id,
        MAX(fld.actual_delivery_date) AS actual_delivery_date
    FROM
        doo_fulfill_line_details fld
        INNER JOIN doo_fulfill_lines_all fl
            ON fl.fulfill_line_id = fld.fulfill_line_id
            AND fl.shippable_flag = 'Y'
        INNER JOIN doo_headers_all h
            ON h.header_id = fl.header_id
            AND h.order_type_code IN ('WP_B2C', 'WP_B2C_CPU')
    WHERE
        fld.actual_delivery_date IS NOT NULL
        AND (
            h.source_order_number IN (:P_SALES_ORDER_NUMBERS)
            OR (
                :P_FROM_DATE IS NOT NULL
                AND :P_TO_DATE IS NOT NULL
                AND fld.actual_delivery_date >= CAST(:P_FROM_DATE AS DATE)
                AND fld.actual_delivery_date <= CAST(:P_TO_DATE AS DATE)
            )
        )
    GROUP BY
        fld.fulfill_line_id
),
rmcs_lines AS (
    SELECT
        fl.fulfill_line_number,
        fl.fulfilled_qty,
        sdl.doc_line_id_int_1,
        sdl.doc_line_id_int_2,
        sdl.doc_line_id_int_3,
        sdl.doc_line_id_int_4,
        sdl.doc_line_id_int_5,
        sdl.doc_line_id_char_1,
        sdl.doc_line_id_char_2,
        sdl.doc_line_id_char_3,
        sdl.doc_line_id_char_4,
        sdl.doc_line_id_char_5,
        TO_NUMBER(
            TO_CHAR(pol.customer_contract_header_id)
            || TO_CHAR(pol.document_line_id)
        ) AS doc_addl_sub_line_id_int_1,
        CASE
            WHEN h.order_type_code IN ('WP_B2C', 'WP_B2C_CPU')
                THEN fld.actual_delivery_date
            WHEN h.order_type_code IN ('WP_B2C_TAKEAWAY', 'WP_B2B_WHOLESALE')
                THEN fl.fulfillment_date
        END AS fulfillment_date
    FROM
        doo_headers_all h
        INNER JOIN doo_lines_all l
            ON l.header_id = h.header_id
        INNER JOIN doo_fulfill_lines_all fl
            ON fl.header_id = h.header_id
            AND fl.line_id = l.line_id
            AND fl.shippable_flag = 'Y'
        LEFT JOIN fulfill_delivery fld
            ON fld.fulfill_line_id = fl.fulfill_line_id
        -- Related source document contract already exists in RMCS
        INNER JOIN vrm_source_doc_lines sdl
            ON sdl.doc_line_id_char_1 = h.source_order_number
            AND sdl.doc_line_id_int_1 = l.line_id
            AND sdl.doc_line_id_int_2 = CASE
                -- INT_2 stores LINE_NUMBER for normal lines; decimal FULFILL_LINE_NUMBER for split shipments (e.g. 1.1).
                WHEN fl.fulfill_line_number != TRUNC(fl.fulfill_line_number)
                    THEN fl.fulfill_line_number
                ELSE l.line_number
            END
            AND sdl.source_system = 'Warby'
            AND sdl.document_type_code = 'Warby'
        -- Line is on a contract: active performance obligation for that source doc line
        INNER JOIN vrm_perf_obligation_lines pol
            ON pol.document_line_id = sdl.document_line_id
            AND NVL(pol.removed_flag, 'N') = 'N'
    WHERE
        h.order_type_code IN (
            'WP_B2C',
            'WP_B2C_CPU',
            'WP_B2C_TAKEAWAY',
            'WP_B2B_WHOLESALE'
        )
        AND fl.status_code IN (
            'SHIPPED',
            'AWAIT_BILLING',
            'BILLED'
        )
        -- AND NVL(fl.fulfilled_qty, 0) > 0
        -- Line not already in RMCS as a proof of delivery line
        AND NOT EXISTS (
            SELECT
                1
            FROM
                vrm_source_doc_addl_sublines ads
            WHERE
                ads.document_line_id = sdl.document_line_id
                AND ads.doc_additional_sline_id_int_1 = TO_NUMBER(
                    TO_CHAR(pol.customer_contract_header_id)
                    || TO_CHAR(pol.document_line_id)
                )
                AND ads.source_system = 'Warby'
                AND ads.document_type_code = 'Warby'
                AND UPPER(ads.additional_se_type_code) LIKE '%PROOF%DELIVERY%'
        )
        AND (
            CASE
                WHEN h.order_type_code IN ('WP_B2C', 'WP_B2C_CPU')
                    THEN fld.actual_delivery_date
                WHEN h.order_type_code IN ('WP_B2C_TAKEAWAY', 'WP_B2B_WHOLESALE')
                    THEN fl.fulfillment_date
            END IS NOT NULL
        )
        AND (
            CASE
                WHEN h.order_type_code IN ('WP_B2C', 'WP_B2C_CPU')
                    THEN fld.actual_delivery_date
                WHEN h.order_type_code = 'WP_B2C_TAKEAWAY'
                    THEN fl.fulfillment_date
                WHEN h.order_type_code = 'WP_B2B_WHOLESALE'
                    THEN fl.actual_ship_date
            END IS NOT NULL
        )
        AND (
            (
                :P_FROM_DATE IS NOT NULL
                AND :P_TO_DATE IS NOT NULL
                AND CASE
                    WHEN h.order_type_code IN ('WP_B2C', 'WP_B2C_CPU')
                        THEN fld.actual_delivery_date
                    WHEN h.order_type_code = 'WP_B2C_TAKEAWAY'
                        THEN fl.fulfillment_date
                    WHEN h.order_type_code = 'WP_B2B_WHOLESALE'
                        THEN fl.actual_ship_date
                END >= CAST(:P_FROM_DATE AS DATE)
                AND CASE
                    WHEN h.order_type_code IN ('WP_B2C', 'WP_B2C_CPU')
                        THEN fld.actual_delivery_date
                    WHEN h.order_type_code = 'WP_B2C_TAKEAWAY'
                        THEN fl.fulfillment_date
                    WHEN h.order_type_code = 'WP_B2B_WHOLESALE'
                        THEN fl.actual_ship_date
                END <= CAST(:P_TO_DATE AS DATE)
            )
            OR h.source_order_number IN (:P_SALES_ORDER_NUMBERS)
        )
)
SELECT
    'A'                                            AS record_type,
    'Warby'                                        AS document_type_code,
    rl.doc_addl_sub_line_id_int_1,
    CAST(NULL AS NUMBER)                           AS doc_addl_sub_line_id_int_2,
    CAST(NULL AS NUMBER)                           AS doc_addl_sub_line_id_int_3,
    CAST(NULL AS NUMBER)                           AS doc_addl_sub_line_id_int_4,
    CAST(NULL AS NUMBER)                           AS doc_addl_sub_line_id_int_5,
    CAST(NULL AS VARCHAR2(30))                     AS doc_sub_line_id_char_1,
    CAST(NULL AS VARCHAR2(30))                     AS doc_sub_line_id_char_2,
    CAST(NULL AS VARCHAR2(30))                     AS doc_sub_line_id_char_3,
    CAST(NULL AS VARCHAR2(30))                     AS doc_sub_line_id_char_4,
    CAST(NULL AS VARCHAR2(30))                     AS doc_sub_line_id_char_5,
    'Warby'                                        AS source_system,
    rl.doc_line_id_int_1,
    rl.doc_line_id_int_2,
    rl.doc_line_id_int_3,
    rl.doc_line_id_int_4,
    rl.doc_line_id_int_5,
    rl.doc_line_id_char_1,
    rl.doc_line_id_char_2,
    rl.doc_line_id_char_3,
    rl.doc_line_id_char_4,
    rl.doc_line_id_char_5,
    'Proof of delivery'                            AS additional_satisfaction_event,
    rl.fulfilled_qty                               AS satisfied_quantity,
    CAST(NULL AS NUMBER)                           AS satisfied_percent,
    CAST(NULL AS NUMBER)                           AS amount_applied,
    rl.fulfillment_date
FROM
    rmcs_lines rl
ORDER BY
    rl.doc_line_id_char_1,
    rl.doc_line_id_int_2,
    rl.fulfill_line_number

--
-- Order types (FND_LOOKUP_VALUES_VL, ORA_DOO_ORDER_TYPES):
--   WP_B2C            B2C                    -> delivery date (shipping) for POD
--   WP_B2C_CPU        B2C CPU                -> delivery date (shipping) for POD
--   WP_B2C_TAKEAWAY   B2C Customer Takeaway  -> fulfillment date for POD
--   WP_B2B_WHOLESALE  B2B/Wholesale Orders   -> fulfillment date for POD; ship date for filter
-- Excluded: WP_B2B_RETURN, WP_B2C_RETURN, WP_ORDER
--
-- RMCS contract exists: inner join to VRM_SOURCE_DOC_LINES + VRM_PERF_OBLIGATION_LINES
--   (OM rows without a matching RMCS source doc line are excluded).
-- RMCS source doc line keys (validated against POC062):
--   DOC_LINE_ID_CHAR_1 = SOURCE_ORDER_NUMBER
--   DOC_LINE_ID_INT_1  = LINE_ID
--   DOC_LINE_ID_INT_2  = LINE_NUMBER; FULFILL_LINE_NUMBER when split (e.g. 1.1)
--
-- Fulfill line status (DOO_STATUSES_VL):
--   SHIPPED       Shipped
--   AWAIT_BILLING Awaiting Billing
--   BILLED        Billed

-- Actual delivery date (shipping) for WP_B2C / WP_B2C_CPU only.
-- Parameter scope matches rmcs_lines (SO list or delivery-date range).
WITH fulfill_delivery AS (
    SELECT
        fld.fulfill_line_id,
        MAX(fld.actual_delivery_date) AS actual_delivery_date
    FROM
        doo_fulfill_line_details fld

                AND fld.actual_delivery_date <= CAST(:P_TO_DATE AS DATE)
            )
        )
    GROUP BY
        fld.fulfill_line_id
),
rmcs_lines AS (
    SELECT
        h.header_id,
        h.source_order_number                          AS sales_order_number,
        h.order_type_code,
        l.line_id,
        l.line_number,
        fl.fulfill_line_id,
        fl.fulfill_line_number,
        fl.status_code,
        fl.actual_ship_date,
        fl.fulfillment_date,
        fl.fulfilled_qty,
        fl.last_update_date AS fulfill_line_last_update_date,
        fld.actual_delivery_date,
        sdl.document_line_id,
        sdl.doc_line_id_int_1,
        sdl.doc_line_id_int_2,
        sdl.doc_line_id_int_3,
        sdl.doc_line_id_int_4,
        sdl.doc_line_id_int_5,
        sdl.doc_line_id_char_1,
        sdl.doc_line_id_char_2,
        sdl.doc_line_id_char_3,
        sdl.doc_line_id_char_4,
        sdl.doc_line_id_char_5,
        pol.customer_contract_header_id,
        pol.document_line_id AS perf_obligation_document_line_id,
        TO_NUMBER(
            TO_CHAR(pol.customer_contract_header_id)
            || TO_CHAR(pol.document_line_id)
        ) AS doc_addl_sub_line_id_int_1,
        CASE
            WHEN h.order_type_code IN ('WP_B2C', 'WP_B2C_CPU')
                THEN fld.actual_delivery_date
            WHEN h.order_type_code IN ('WP_B2C_TAKEAWAY', 'WP_B2B_WHOLESALE')

            WHEN h.order_type_code = 'WP_B2C_TAKEAWAY'
                THEN fl.fulfillment_date
            WHEN h.order_type_code = 'WP_B2B_WHOLESALE'
                THEN fl.actual_ship_date
        END AS extraction_trigger_date
    FROM
        doo_headers_all h
        INNER JOIN doo_lines_all l
            ON l.header_id = h.header_id
        INNER JOIN doo_fulfill_lines_all fl
            ON fl.header_id = h.header_id
            AND fl.line_id = l.line_id
            AND fl.shippable_flag = 'Y'
        LEFT JOIN fulfill_delivery fld
            ON fld.fulfill_line_id = fl.fulfill_line_id
        -- Related source document contract already exists in RMCS
        INNER JOIN vrm_source_doc_lines sdl

            AND sdl.doc_line_id_int_1 = l.line_id
            AND sdl.doc_line_id_int_2 = CASE
                -- INT_2 stores LINE_NUMBER for normal lines; decimal FULFILL_LINE_NUMBER for split shipments (e.g. 1.1).
                WHEN fl.fulfill_line_number != TRUNC(fl.fulfill_line_number)
                    THEN fl.fulfill_line_number
                ELSE l.line_number
            END

            AND sdl.document_type_code = 'Warby'
        -- Line is on a contract: active performance obligation for that source doc line
        INNER JOIN vrm_perf_obligation_lines pol
