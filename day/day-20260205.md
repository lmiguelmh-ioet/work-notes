- DLQ Strategies for Topics
	- SQS forward: SNS -> SQS -> DLQ
	- How it works:
	- 1. SNS immediately delivers to SQS (push)
	- 2. Message sits in SQS until consumer processes it
	- 3. If consumer fails, message stays in queue (visibility timeout)
	- 4. Consumer retries until success or DLQ redrive
- SNS cannot natively restrict subscriber types, but you can enforce policies:
	- IAM Policy Restrictions `Condition > StringNotEquals > sns:Protocol: sqs / sns:Endpoint : *.fifo`
	- Custom lambda triggered by CloudTrail logs on Sns:subscribe 
		- sns:subscribe -> cloudtraile vent -> cloudwatch event -> lambda (remediate (unsusbscibe))
- "Manual" DLQ implementation
	- endpoint logic on each subscriber
	- secondary SNS topic as "DLQ"
		- THIS HANDLES THE CASE WHERE MESSAGE REACH SNS
		- sns retries handles the case where message reach subscriptors!
PUBLISH FAILURES:
   SNS Client → SNS Service (Network/Validation/Permission errors)
   │
   ├── Network timeout
   ├── Invalid message (too large, invalid attributes)
   ├── Permission denied (sns:Publish)
   ├── Topic doesn't exist
   └── Throttling (exceeded limits)
DELIVERY FAILURES (What SNS retries handle):
   SNS Service → Subscriber Endpoint
   │
   ├── HTTP 5xx errors
   ├── Lambda exceptions
   ├── SQS temporarily unavailable
   └── Network issues to subscriber

        for attempt in range(self.max_retries):
            try:
                # Try to reach SNS service
                response = sns.publish(...)
                return response  # Success!
            except Exception as e:
                if attempt == self.max_retries - 1:
                    # All retries failed to even REACH SNS
                    # Send to DLQ so message isn't lost
                    self._send_to_dlq(message, e)


class SNSWithDLQPublisher:
    def __init__(
        self,
        main_topic_arn: str,
        dlq_topic_arn: str,
        max_retries: int = 3
    ):
        self.main_topic_arn = main_topic_arn
        self.dlq_topic_arn = dlq_topic_arn
        self.max_retries = max_retries
    
    def publish_with_dlq_fallback(
        self,
        message: Dict[str, Any],
        message_attributes: Optional[Dict[str, Any]] = None,
        message_group_id: Optional[str] = None,
        message_deduplication_id: Optional[str] = None
    ) -> Dict[str, Any]:
        """
        Publish to main topic, fallback to DLQ topic on failure
        """
        attempt = 0
        last_error = None
        
        while attempt < self.max_retries:
            try:
                # Prepare publish parameters
                publish_params = {
                    'TopicArn': self.main_topic_arn,
                    'Message': json.dumps(message),
                }
                
                # Add optional parameters
                if message_attributes:
                    publish_params['MessageAttributes'] = message_attributes
                
                # FIFO-specific parameters
                if message_group_id:
                    publish_params['MessageGroupId'] = message_group_id
                if message_deduplication_id:
                    publish_params['MessageDeduplicationId'] = message_deduplication_id
                
                # Attempt publish
                response = sns.publish(**publish_params)
                
                logger.info(f"Successfully published to main topic: {response['MessageId']}")
                return {
                    'success': True,
                    'message_id': response['MessageId'],
                    'topic_arn': self.main_topic_arn,
                    'attempt': attempt + 1
                }
                
            except Exception as e:
                last_error = e
                attempt += 1
                logger.warning(f"Publish attempt {attempt} failed: {str(e)}")
                
                if attempt < self.max_retries:
                    # Exponential backoff before retry
                    import time
                    time.sleep(2 ** attempt)
        
        # All retries failed, send to DLQ
        logger.error(f"All {self.max_retries} attempts failed. Sending to DLQ.")
        return self._publish_to_dlq(message, last_error, message_attributes)
    
    def _publish_to_dlq(
        self,
        original_message: Dict[str, Any],
        error: Exception,
        message_attributes: Optional[Dict[str, Any]] = None
    ) -> Dict[str, Any]:
        """Publish failure details to DLQ topic"""
        
        dlq_message = {
            'type': 'SNS_PUBLISH_FAILURE',
            'timestamp': datetime.utcnow().isoformat(),
            'original_message': original_message,
            'error': {
                'type': error.__class__.__name__,
                'message': str(error),
                'details': self._extract_error_details(error)
            },
            'metadata': {
                'main_topic_arn': self.main_topic_arn,
                'dlq_topic_arn': self.dlq_topic_arn,
                'retry_attempts': self.max_retries
            }
        }
        
        # Add original message attributes if available
        dlq_attributes = {}
        if message_attributes:
            # Prefix DLQ attributes to avoid conflicts
            for key, value in message_attributes.items():
                dlq_attributes[f'dlq_original_{key}'] = value
        
        # Add failure metadata as message attributes for filtering
        dlq_attributes['failure_type'] = {
            'DataType': 'String',
            'StringValue': 'sns_publish_failure'
        }
        dlq_attributes['error_code'] = {
            'DataType': 'String',
            'StringValue': error.__class__.__name__
        }
        
        try:
            response = sns.publish(
                TopicArn=self.dlq_topic_arn,
                Message=json.dumps(dlq_message),
                MessageAttributes=dlq_attributes
            )
            
            logger.info(f"Published to DLQ topic: {response['MessageId']}")
            return {
                'success': True,
                'message_id': response['MessageId'],
                'topic_arn': self.dlq_topic_arn,
                'dlq_published': True,
                'error': str(error)
            }
            
        except Exception as dlq_error:
            # Critical failure - DLQ publish also failed
            logger.critical(f"DLQ publish failed: {str(dlq_error)}")
            
            # Last resort: Log to CloudWatch
            import traceback
            logger.critical({
                'critical_failure': 'DLQ_UNAVAILABLE',
                'original_message': original_message,
                'sns_error': str(error),
                'dlq_error': str(dlq_error),
                'traceback': traceback.format_exc()
            })
            
            return {
                'success': False,
                'dlq_published': False,
                'errors': {
                    'sns_error': str(error),
                    'dlq_error': str(dlq_error)
                }
            }
    
    def _extract_error_details(self, error: Exception) -> Dict[str, Any]:
        """Extract structured error details"""
        error_details = {
            'error_class': error.__class__.__name__,
            'error_message': str(error)
        }
        
        # Check for specific AWS error codes
        if hasattr(error, 'response'):
            error_details['aws_response'] = error.response
            
            # Extract useful info from AWS error response
            if 'Error' in error.response:
                error_details['aws_error_code'] = error.response['Error'].get('Code')
                error_details['aws_error_message'] = error.response['Error'].get('Message')
        
        return error_details


class SNSWithDLQ:
    def __init__(self, main_topic_arn, dlq_topic_arn):
        self.main_topic = main_topic_arn
        self.dlq_topic = dlq_topic_arn
    
    async def publish_with_dlq(self, message):
        try:
            # Try main topic
            response = await sns.publish(
                TopicArn=self.main_topic,
                Message=message
            )
            return response
        except Exception as e:
            # Publish failure to DLQ topic
            await sns.publish(
                TopicArn=self.dlq_topic,
                Message=json.dumps({
                    "error": str(e),
                    "original_message": message,
                    "timestamp": datetime.utcnow().isoformat()
                })
            )
            raise
