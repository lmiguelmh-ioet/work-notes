### Query to get unique suppliers
```
SELECT DISTINCT
    hp.party_id,
    hp.party_name AS supplier_name
FROM
    po_headers_all pha
    JOIN poz_suppliers psv ON psv.vendor_id = pha.vendor_id
    JOIN hz_parties hp ON hp.party_id = psv.party_id
    JOIN po_lines_all pla ON pla.po_header_id = pha.po_header_id
        AND (pla.cancel_flag IS NULL OR pla.cancel_flag != 'Y')
    JOIN po_line_locations_all plla ON pla.po_header_id = plla.po_header_id
        AND pla.po_line_id = plla.po_line_id
        AND plla.schedule_status != 'CANCELED'
        AND (plla.cancel_flag IS NULL OR plla.cancel_flag != 'Y')
        AND pha.attribute1 NOT IN ('LMS', 'Uncuts', 'SF Conversion')
    JOIN egp_system_items_b esi ON esi.inventory_item_id = pla.item_id
        AND esi.organization_id = plla.ship_to_organization_id
        AND esi.ACD_TYPE = 'PROD'
    JOIN egp_item_classes_b eic ON esi.item_catalog_group_id = eic.item_class_id
WHERE
    1 = 1
    AND (pla.cancel_flag IS NULL OR pla.cancel_flag != 'Y')
    AND (plla.cancel_flag IS NULL OR plla.cancel_flag != 'Y')
    AND pha.attribute1 NOT IN ('LMS', 'Uncuts', 'SF Conversion')
    AND hp.party_name IN ('Gatto Asia Limited', 'Case Optics LLC', 'Menicon America Inc')
ORDER BY
    hp.party_name;
```

### Query to get item class codes
```
SELECT DISTINCT
    eic.item_class_id,
    eic.item_class_code,
    eic.item_class_name
FROM
    po_headers_all pha
    JOIN po_lines_all pla ON pla.po_header_id = pha.po_header_id
        AND (pla.cancel_flag IS NULL OR pla.cancel_flag != 'Y')
    JOIN po_line_locations_all plla ON pla.po_header_id = plla.po_header_id
        AND pla.po_line_id = plla.po_line_id
        AND plla.schedule_status != 'CANCELED'
        AND (plla.cancel_flag IS NULL OR plla.cancel_flag != 'Y')
        AND pha.attribute1 NOT IN ('LMS', 'Uncuts', 'SF Conversion')
    JOIN egp_system_items_b esi ON esi.inventory_item_id = pla.item_id
        AND esi.organization_id = plla.ship_to_organization_id
        AND esi.ACD_TYPE = 'PROD'
    JOIN egp_item_classes_b eic_b ON esi.item_catalog_group_id = eic_b.item_class_id
    JOIN egp_item_classes_vl eic ON eic.item_class_id = eic_b.item_class_id
WHERE
    1 = 1
    AND (pla.cancel_flag IS NULL OR pla.cancel_flag != 'Y')
    AND (plla.cancel_flag IS NULL OR plla.cancel_flag != 'Y')
    AND pha.attribute1 NOT IN ('LMS', 'Uncuts', 'SF Conversion')
    AND eic.item_class_code IN ('Frames', 'Accessories')
ORDER BY
    eic.item_class_code
```
