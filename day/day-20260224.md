@dataclass(frozen=True)
class _StripeRow(_Row):
    @property
    def transaction_code(self) -> str:
        val = self.payout_file.get_value(
            self.row_index, "reporting_category"
        ).strip().lower()
        if val == "charge":
            return Constants.TRANSACTION_CODE_CHARGE_OR_RECEIPT
        if val in ("refund", "partial_capture_reversal"):
            return Constants.TRANSACTION_CODE_REFUND
        raise BankStatementFbdiMapperError(
            f"Unknown transaction code for: {PayoutProvider.STRIPE.value} and reporting_category: {val}"

            self.row_index, "automatic_payout_effective_at"
        )
        dt = datetime.strptime(raw.strip(), Constants.PAYOUT_STRIPE_DATETIME_FORMAT)
        return dt.replace(tzinfo=timezone.utc)

    def structured_reference(self) -> str:
        return self.payout_file.get_value(self.row_index, "automatic_payout_id")

        return f"{amount_value}"

    def booking_date(self) -> datetime:
        raw = self.payout_file.get_value(
            self.row_index, "Transaction Completion Date"

        return dt.astimezone(timezone.utc)

        return ""

        if is_receipt:
            return self.payout_file.get_value(self.row_index, "Transaction ID")
        return self.payout_file.get_value(self.row_index, "Paypal Reference ID")

class BankStatementFbdiMapper:
    @staticmethod
    def _get_bank_account_number(prefix: CountryPrefix, provider: PayoutProvider) -> str:
        key = (provider, prefix)
        if key not in Constants.BANK_ACCOUNT_NUMBER_MAP:
            raise BankStatementFbdiMapperError(
                f"Unknown bank account number for: {prefix.value}-{provider.value}"
            )
        return Constants.BANK_ACCOUNT_NUMBER_MAP[key]

    @staticmethod
    def _get_currency(prefix: CountryPrefix) -> str:
        if prefix not in Constants.CURRENCY_MAP:
            raise BankStatementFbdiMapperError(
                f"Unknown currency code for: {prefix.value}"
            )
        return Constants.CURRENCY_MAP[prefix]

    @staticmethod
    def _get_statement_number(metadata: PayoutFilenameMetadata) -> str:
        date_formatted = metadata.date.strftime(Constants.FBDI_STATEMENT_NUMBER_DATE_FORMAT)
        return f"{metadata.prefix.value}-{metadata.provider.value}-{date_formatted}"

    def _build_headers_csv(
        self,
        statement_number: str,
        bank_account_number: str,
        currency_code: str,
        statement_date: str,
    ) -> BankStatementHeadersCsv:
        header_columns = [
            "Statement Number",
            "Bank Account Number",
            "Intra-Day Statement",
            "Statement Date",
            "Currency Code",
            "Statement From Date",
            "Statement To Date",
        ]
        data_row = [
            statement_number,
            bank_account_number,
            Constants.STATEMENT_INTRA_DAY,
            statement_date,
            currency_code,
            statement_date,
            statement_date,
        ]
        return BankStatementHeadersCsv(header=header_columns, row=data_row)

    def _build_lines_csv(
        self,
        metadata: PayoutFilenameMetadata,
        statement_number: str,
        bank_account_number: str,
        currency_code: str,
        payout_file: PayoutFile,
    ) -> BankStatementLinesCsv:
        line_columns = [
            "Statement Number",
            "Bank Account Number",
            "Line Number",
            "Transaction Code",
            "Transaction Type",
            "Amount",
            "Bank Account Currency",
            "Booking Date",
            "Structured Payment Reference",
            "Reconciliation Reference",
            "Credit Debit Indicator",
        ]

        strategy = _get_provider_strategy(metadata.provider)

        def build_line(
            line_number: int,
            row_index: int,
        ) -> List[str]:
            provider_row = strategy.build_row(payout_file, row_index)
            booking_date = provider_row.booking_date.strftime(
                Constants.FBDI_DATE_FORMAT
            )

            return [
                statement_number,
                bank_account_number,
                str(line_number),
                provider_row.transaction_code,
                Constants.TRANSACTION_TYPE_ACH,
                provider_row.amount,
                currency_code,
                booking_date,
                provider_row.structured_reference,
                provider_row.reconciliation_reference,
                provider_row.credit_debit,
            ]

        lines: List[List[str]] = []
        for row_index in range(len(payout_file.rows)):
            line_number = row_index + 1
            lines.append(build_line(line_number, row_index))

        return BankStatementLinesCsv(header=line_columns, rows=lines)

    def to_fbdi_csv(
        self,
        metadata: PayoutFilenameMetadata,
        payout_file: PayoutFile,
    ) -> Tuple[BankStatementHeadersCsv, BankStatementLinesCsv]:
        if not payout_file.rows:
            raise BankStatementFbdiMapperError("No data rows found in payout file.")

        statement_number = self._get_statement_number(metadata)
        bank_account_number = self._get_bank_account_number(
            prefix=metadata.prefix,
            provider=metadata.provider,
        )
        strategy = _get_provider_strategy(metadata.provider)
        statement_date_dt = strategy.get_statement_date(payout_file)
        statement_date = statement_date_dt.strftime(
            Constants.FBDI_DATE_FORMAT
        )
        currency_code = self._get_currency(prefix=metadata.prefix)

        headers_csv = self._build_headers_csv(
            statement_number=statement_number,
            bank_account_number=bank_account_number,
            currency_code=currency_code,
            statement_date=statement_date,
        )

        lines_csv = self._build_lines_csv(
            metadata=metadata,
            statement_number=statement_number,
            bank_account_number=bank_account_number,
            currency_code=currency_code,
            payout_file=payout_file,
        )
        return headers_csv, lines_csv

            dt = datetime.strptime(raw.strip(), Constants.PAYOUT_STRIPE_DATETIME_FORMAT)
            return dt.replace(tzinfo=timezone.utc)
        if provider is PayoutProvider.PAYPAL:
            raw = payout_file.get_value(0, "Transaction Completion Date")
            dt = datetime.strptime(raw.strip(), Constants.PAYOUT_PAYPAL_DATETIME_FORMAT)
            return dt.astimezone(timezone.utc)
        if provider is PayoutProvider.AFFIRM:
            raw = payout_file.get_value(0, "date")
            dt = datetime.strptime(raw.strip(), Constants.PAYOUT_AFFIRM_DATETIME_FORMAT)
            return dt.replace(tzinfo=timezone.utc)
        raise BankStatementFbdiMapperError(f"Unknown statement date for: {provider.value}")

    @staticmethod
    def _get_transaction_code(
        provider: PayoutProvider,
        payout_file: PayoutFile,
        row_index: int,
    ) -> str:
        """Return 165 (charge/receipt) or 455 (refund) from provider-specific column."""
        if provider is PayoutProvider.STRIPE:
            val = payout_file.get_value(row_index, "reporting_category").strip().lower()
            if val == "charge":
                return Constants.TRANSACTION_CODE_CHARGE_OR_RECEIPT
            if val in ("refund", "partial_capture_reversal"):
                return Constants.TRANSACTION_CODE_REFUND
            raise BankStatementFbdiMapperError(
                f"Unknown transaction code for: {provider.value} and reporting_category: {val}"
            )

        elif provider is PayoutProvider.PAYPAL:
            val = payout_file.get_value(
                row_index, "Transaction Event Code"
            ).strip()
            if val == "T0003":
                return Constants.TRANSACTION_CODE_CHARGE_OR_RECEIPT
            if val == "T1107":
                return Constants.TRANSACTION_CODE_REFUND
            raise BankStatementFbdiMapperError(
                f"Unknown transaction code for: {provider.value} and Transaction Event Code: {val}"
            )

        elif provider is PayoutProvider.AFFIRM:
            val = payout_file.get_value(row_index, "event_type").strip()
            if val == "loan_captured":
                return Constants.TRANSACTION_CODE_CHARGE_OR_RECEIPT
            if val == "loan_refunded":
                return Constants.TRANSACTION_CODE_REFUND
            raise BankStatementFbdiMapperError(
                f"Unknown transaction code for: {provider.value} and event_type: {val}"
            )
        else:
            raise BankStatementFbdiMapperError(
                f"Unknown transaction code for: {provider.value}"
            )

    @staticmethod
    def _get_amount(
        provider: PayoutProvider,
        payout_file: PayoutFile,
        row_index: int,
    ) -> str:
        """Return provider-specific amount as a string."""
        if provider is PayoutProvider.STRIPE:
            return payout_file.get_value(row_index, "net")

        if provider is PayoutProvider.PAYPAL:
            gross_amount = payout_file.get_value(
                row_index, "Gross Transaction Amount"
            )
            try:
                amount_value = float(gross_amount) / 100
            except ValueError as error:
                raise BankStatementFbdiMapperError(
                    f"Invalid Gross Transaction Amount for: {provider.value}: {gross_amount}"
                ) from error
            return f"{amount_value}"

        if provider is PayoutProvider.AFFIRM:
            event_type = payout_file.get_value(row_index, "event_type")
            if event_type == "loan_captured":
                return payout_file.get_value(row_index, "Sales")
            if event_type == "loan_refunded":
                return payout_file.get_value(row_index, "Refunds")
            raise BankStatementFbdiMapperError(
                f"Unknown amount mapping for: {provider.value} and event_type: {event_type}"
            )

        raise BankStatementFbdiMapperError(
            f"Unknown amount mapping for: {provider.value}"
        )

    @staticmethod
    def _get_booking_date(
        provider: PayoutProvider,
        payout_file: PayoutFile,
        row_index: int,
    ) -> datetime:
        """Parse provider-specific booking/transaction date from the given row."""
        if provider is PayoutProvider.STRIPE:
            raw = payout_file.get_value(row_index, "automatic_payout_effective_at")
            dt = datetime.strptime(raw.strip(), Constants.PAYOUT_STRIPE_DATETIME_FORMAT)
            return dt.replace(tzinfo=timezone.utc)

        elif provider is PayoutProvider.PAYPAL:
            raw = payout_file.get_value(row_index, "Transaction Completion Date")
            dt = datetime.strptime(raw.strip(), Constants.PAYOUT_PAYPAL_DATETIME_FORMAT)
            return dt.astimezone(timezone.utc)

        elif provider is PayoutProvider.AFFIRM:
            raw = payout_file.get_value(row_index, "date")
            dt = datetime.strptime(raw.strip(), Constants.PAYOUT_AFFIRM_DATETIME_FORMAT)
            return dt.replace(tzinfo=timezone.utc)

        raise BankStatementFbdiMapperError(
            f"Unknown booking date for provider: {provider.value}"
        )

    @staticmethod
    def _get_structured_reference(
        provider: PayoutProvider,
        payout_file: PayoutFile,
        row_index: int,
    ) -> str:
        """Return provider-specific structured payment reference."""
        if provider is PayoutProvider.STRIPE:
            return payout_file.get_value(row_index, "automatic_payout_id")

        if provider is PayoutProvider.PAYPAL:
            return ""

        if provider is PayoutProvider.AFFIRM:
            return payout_file.get_value(row_index, "deposit_id")

        raise BankStatementFbdiMapperError(
            f"Unknown structured reference mapping for provider: {provider.value}"
        )

    @staticmethod
    def _get_reconciliation_reference(
        provider: PayoutProvider,
        payout_file: PayoutFile,
        row_index: int,
    ) -> str:
        """Return provider-specific reconciliation reference."""
        transaction_code = BankStatementFbdiMapper._get_transaction_code(
            provider, payout_file, row_index
        )
        is_receipt = (
            transaction_code == Constants.TRANSACTION_CODE_CHARGE_OR_RECEIPT
        )

        if provider is PayoutProvider.STRIPE:
            return payout_file.get_value(row_index, "Charge id")

        if provider is PayoutProvider.PAYPAL:
            if is_receipt:
                return payout_file.get_value(row_index, "Transaction ID")
            return payout_file.get_value(row_index, "Paypal Reference ID")

        if provider is PayoutProvider.AFFIRM:
            return payout_file.get_value(row_index, "Affirm_Charge_id")

        raise BankStatementFbdiMapperError(
            f"Unknown reconciliation reference mapping for provider: {provider.value}"
        )

    @staticmethod
    def _get_credit_debit_indicator(
        provider: PayoutProvider,
        payout_file: PayoutFile,
        row_index: int,
    ) -> str:
        """Return CRDT/DBIT based on transaction code."""
        transaction_code = BankStatementFbdiMapper._get_transaction_code(
            provider, payout_file, row_index
        )
        if transaction_code == Constants.TRANSACTION_CODE_CHARGE_OR_RECEIPT:
            return Constants.CREDIT_DEBIT_CREDIT
        if transaction_code == Constants.TRANSACTION_CODE_REFUND:
            return Constants.CREDIT_DEBIT_DEBIT
        raise BankStatementFbdiMapperError(
            f"Unknown credit/debit indicator for transaction code: {transaction_code}"
        )

    def _build_headers_csv(
        self,
        statement_number: str,
        bank_account_number: str,
        currency_code: str,
        statement_date: str,
    ) -> BankStatementHeadersCsv:
        header_columns = [
            "Statement Number",
            "Bank Account Number",
            "Intra-Day Statement",
            "Statement Date",
            "Currency Code",
            "Statement From Date",
            "Statement To Date",
        ]
        data_row = [
            statement_number,
            bank_account_number,
            Constants.STATEMENT_INTRA_DAY,
            statement_date,
            currency_code,
            statement_date,
            statement_date,
        ]
        return BankStatementHeadersCsv(header=header_columns, row=data_row)

    def _build_lines_csv(
        self,
        metadata: PayoutFilenameMetadata,
        statement_number: str,
        bank_account_number: str,
        currency_code: str,
        payout_file: PayoutFile,
    ) -> BankStatementLinesCsv:
        line_columns = [
            "Statement Number",
            "Bank Account Number",
            "Line Number",
            "Transaction Code",
            "Transaction Type",
            "Amount",
            "Bank Account Currency",
            "Booking Date",
            "Structured Payment Reference",
            "Reconciliation Reference",
            "Credit Debit Indicator",
        ]

        strategy = _get_provider_strategy(metadata.provider)

        def build_line(
            line_number: int,
            row_index: int,
        ) -> List[str]:
            provider_row = strategy.build_row(payout_file, row_index)
            booking_date = provider_row.booking_date.strftime(
                Constants.FBDI_DATE_FORMAT
            )

            return [
                statement_number,
                bank_account_number,
                str(line_number),
                provider_row.transaction_code,
                Constants.TRANSACTION_TYPE_ACH,
                provider_row.amount,
                currency_code,
                booking_date,
                provider_row.structured_reference,
                provider_row.reconciliation_reference,
                provider_row.credit_debit,
            ]

        lines: List[List[str]] = []
        for row_index in range(len(payout_file.rows)):
            line_number = row_index + 1
            lines.append(build_line(line_number, row_index))

        return BankStatementLinesCsv(header=line_columns, rows=lines)

    def to_fbdi_csv(
        self,
        metadata: PayoutFilenameMetadata,
        payout_file: PayoutFile,
    ) -> Tuple[BankStatementHeadersCsv, BankStatementLinesCsv]:
        if not payout_file.rows:
            raise BankStatementFbdiMapperError("No data rows found in payout file.")

        statement_number = self._get_statement_number(metadata)
        bank_account_number = self._get_bank_account_number(
            prefix=metadata.prefix,
            provider=metadata.provider,
        )
        strategy = _get_provider_strategy(metadata.provider)
        statement_date_dt = strategy.get_statement_date(payout_file)
        statement_date = statement_date_dt.strftime(
            Constants.FBDI_DATE_FORMAT
        )
        currency_code = self._get_currency(prefix=metadata.prefix)

        headers_csv = self._build_headers_csv(
            statement_number=statement_number,
            bank_account_number=bank_account_number,
            currency_code=currency_code,
            statement_date=statement_date,
        )

        lines_csv = self._build_lines_csv(
            metadata=metadata,
            statement_number=statement_number,
            bank_account_number=bank_account_number,
            currency_code=currency_code,
            payout_file=payout_file,
        )
        return headers_csv, lines_csv
