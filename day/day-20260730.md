SELECT
  org.name AS business_unit,
  cr.receipt_number AS receipt_number,
  'Refund' AS receivable_trx_name,
  CASE
    WHEN NOT EXISTS (
      SELECT 1
      FROM doo_headers_all dh3
        ,doo_lines_all dl3
        ,doo_fulfill_lines_all dfl3
      WHERE dh3.header_id = dl3.header_id
        AND dl3.header_id = dfl3.header_id
        AND dl3.line_id = dfl3.line_id
        AND dh3.order_number = remit.invoice_reference
        AND dh3.status_code <> 'DOO_REFERENCE'
        AND (
          dh3.status_code <> 'CANCELED'
          OR dfl3.status_code <> 'CANCELED'
        )
    )
    THEN remit.amount_applied
    ELSE NULL                             -- for partial canceled add the amount logic here
  END AS amount_applied,
  TO_CHAR(SYSDATE, 'YYYY-MM-DD') AS apply_date,
  TO_CHAR(SYSDATE, 'YYYY-MM-DD') AS gl_date,
  cr.creation_date AS creation_date,
  '-8' AS applied_payment_schedule_id,
  CASE
    WHEN rm.name = 'CHASUS-0257-Check' THEN 'POE CHECK'
    ELSE 'AR REFUND'
  END AS Payment_Method_Code,
  CASE
    WHEN rm.name = 'CHASUS-0257-Check' THEN 'POE CHECK'
    ELSE 'AR REFUND'
  END AS PayGroupLookupCode,
  CASE
    WHEN rm.name = 'CHASUS-0257-Check'
      THEN (
        SELECT dhai.attribute_char2 || ' ' || dhai.attribute_char3
        FROM doo_headers_all dh2
          ,doo_headers_eff_b dhai
        WHERE dhai.header_id = dh2.header_id
          AND dh2.order_number = remit.invoice_reference
          AND dhai.context_code = 'Customer'
      )
    ELSE NULL
  END AS RemitMessage,
  CASE
    WHEN rm.name = 'CHASUS-0257-Check'
      THEN (
        SELECT dhai.attribute_char7 || ', ' || dhai.attribute_char13 || ',' || dhai.attribute_char8 || ', ' || dhai.attribute_char10 || ', ' || dhai.attribute_char9 || ', ' || dhai.attribute_char6
        FROM doo_headers_all dh2
          ,doo_headers_eff_b dhai
        WHERE dhai.header_id = dh2.header_id
          AND dh2.order_number = remit.invoice_reference
          AND dhai.context_code = 'Customer Billing Address'
      )
    ELSE NULL
  END AS RemittanceMessage,
  CASE
    WHEN rm.name = 'CHASUS-0257-Check'
      THEN (
        SELECT dhai.attribute_char7 || ',' || dhai.attribute_char13 || ',' || dhai.attribute_char8 || ',' || dhai.attribute_char10 || ',' || dhai.attribute_char9 || ',' || dhai.attribute_char6
        FROM doo_headers_all dh2
          ,doo_headers_eff_b dhai
        WHERE dhai.header_id = dh2.header_id
          AND dh2.order_number = remit.invoice_reference
          AND dhai.context_code = 'Customer Ship-To'
      )
    ELSE NULL
  END AS RemitingMessage
FROM
  ar_cash_receipts_all cr
  JOIN ar_cash_remit_refs_all remit ON cr.cash_receipt_id = remit.cash_receipt_id
  JOIN hr_organization_units  org   ON cr.org_id = org.organization_id
  LEFT JOIN ar_receipt_methods rm ON cr.receipt_method_id = rm.receipt_method_id
WHERE
  1 = 1
  AND cr.status = 'UNAPP'
  AND EXISTS (
    SELECT 1
    FROM doo_headers_all dh
    JOIN doo_lines_all dl ON dh.header_id = dl.header_id
                          AND dh.status_code <> 'DOO_REFERENCE'
    JOIN doo_fulfill_lines_all dfl ON dl.header_id = dfl.header_id
                                   AND dl.line_id   = dfl.line_id
                                   AND dfl.status_code = 'CANCELED'
    WHERE remit.invoice_reference = dh.order_number
      AND dh.order_number = '64'
  )
  AND EXISTS (
    SELECT 1
    FROM ar_payment_schedules_all arsch
    where 1 = 1
    and arsch.class = 'PMT'
    and arsch.cash_receipt_id = cr.cash_receipt_id
    and abs(arsch.amount_due_remaining) = cr.amount
    and abs(arsch.amount_due_remaining) > 0
  )
  AND (
    (
      :P_FROM_DATE IS NOT NULL AND :P_TO_DATE IS NOT NULL
      AND cr.creation_date BETWEEN CAST(:P_FROM_DATE AS DATE) AND CAST(:P_TO_DATE AS DATE)
    )
    OR cr.receipt_number IN (:P_RECEIPT_NUMBERS)
  )
